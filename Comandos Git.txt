$ git add pull --> traer de repositorio remoto

$ git remote add origin URL  --> agregar origen remoto de nuestros archivos
$ git remote -v --> ver el origen remoto
$ git remote rm origin --> eliminar origen remoto
$ git push origin HEAD:NombreRama --> enviar rama especifica al repositorio de origen remoto 
$ git pull origin NombreRama --> traer todos los cambios del repositorio de origen remoto

$ git log --all --graph --decorate --oneline --> ver todo el historial del repositorio

$ git show-ref --tags --> ver tags y referencias al commit

$ git stash --> guardar cambios en temporal
$ git stash list --> ver el listado de cambios guardados en temporal

$ git stash drop --> Elimivar cambios guardados en temporal
$ git clean --dry-run --> simulacion de los archivos que van borrar
$ git clean -f --> borra los archivos de la simulacion en fisico

$ git grep palabra --> para buscar una palabra en todo el repositorio
$ git grep -n palabra --> para buscar una palabra en todo el repositorio y mustra la linea
$ git grep -c palabra --> La cantidad de veces que se encuentra la palabra en el archivo
$ git log -S "palabra" --> La cantidad de veces que se encuentra la palabra en los commits
